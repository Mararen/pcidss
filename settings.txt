def evaluar_condiciones(condiciones, perfil):
    for key, value in condiciones.items():
        if perfil.get(key) != value:
            return False
    return True


def prellenar_saq(usuario, entidad, tipo_saq, perfil):
    controles = ControlPCI.objects.filter(
        saq_links__tipo_saq=tipo_saq
    ).distinct()

    for control in controles:
        if perfil["almacena"]:
            respuesta = "NO"
        elif perfil["tercero"] and not perfil["procesa"]:
            respuesta = "IN_PLACE"
        elif perfil["iframe"]:
            respuesta = "IN_PLACE"
        else:
            respuesta = "CCW"

        RespuestaSAQ.objects.update_or_create(
            usuario=usuario,
            entidad=entidad,
            control=control,
            defaults={
                "respuesta": respuesta,
                "comentario": "Prellenado automático por PreTest"
            }
        )

def login_view(request):
    if request.method == "POST":
        username = request.POST.get("username")
        password = request.POST.get("password")
        user = authenticate(request, username=username, password=password)

        if user:
            login(request, user)
            registrar_log(request, "LOGIN", "AUTH", f"Usuario {user.username} inició sesión")
            response = redirect("dashboard")

            if request.POST.get("remember"):
                request.session.set_expiry(60 * 60 * 24 * 30)
                response.set_cookie(
                    "remember_user", username,
                    max_age=60*60*24*30, httponly=True, samesite='Lax'
                )
            return response

        return render(request, "users/login.html", {"error": "Credenciales incorrectas"})

    return render(request, "users/login.html", {
        "remembered_user": request.COOKIES.get("remember_user", "")
    })


@login_required
@permission_required('users.ver_dashboard', raise_exception=True)
def dashboard(request):
    user = request.user
    entidad_activa = Entidad.objects.filter(usuario=user, is_active=True).exists()

    return render(request, "users/dashboard.html", {
        "mod_usuarios":  user.has_perm("users.ver_usuarios"),
        "mod_entidades": user.has_perm("users.ver_entidades"),
        "mod_saq":       user.has_perm("users.ver_saq"),
        "mod_pretest":   user.has_perm("users.usar_pretest"),
        "mod_evidencias":user.has_perm("users.ver_evidencias"),
        "mod_renovacion":user.has_perm("users.gestionar_renovacion"),
        "mod_aoc":       user.has_perm("users.ver_aoc"),
        "entidad_activa": entidad_activa,
    })


def logout_view(request):
    registrar_log(request, "LOGOUT", "AUTH", f"Usuario {request.user.username} cerró sesión")
    logout(request)
    return redirect("login")

@login_required
def saq_dinamico(request):
    entidad = Entidad.objects.filter(usuario=request.user, is_active=True).first()
    if not entidad:
        return redirect("entidades_lista")

    tipo_codigo = request.GET.get("tipo") or request.session.get("saq_recomendado", "A")
    tipo_saq = TipoSAQ.objects.filter(codigo=tipo_codigo).first()

    if not tipo_saq:
        return render(request, "users/saq_dinamico.html", {
            "requisitos": [], "saq_tipo": tipo_codigo,
            "respuestas_dict": {}, "opciones": [],
        })

    controles_validos = ControlPCI.objects.filter(
        saq_links__tipo_saq=tipo_saq
    ).select_related("requisito").distinct()

    if request.method == "POST":
        for key, value in request.POST.items():
            if key.startswith("c_"):
                control_id = key.split("_")[1]
                comentario = request.POST.get(f"comentario_{control_id}", "")
                RespuestaSAQ.objects.update_or_create(
                    usuario=request.user, entidad=entidad, control_id=control_id,
                    defaults={"respuesta": value, "comentario": comentario}
                )
        return redirect("saq_resultado")

    requisitos_dict = {}
    for control in controles_validos:
        req = control.requisito
        if req.id not in requisitos_dict:
            req.controles_filtrados = []
            requisitos_dict[req.id] = req
        requisitos_dict[req.id].controles_filtrados.append(control)

    respuestas_dict = {
        r.control_id: r
        for r in RespuestaSAQ.objects.filter(
            usuario=request.user, entidad=entidad, control__in=controles_validos
        ).select_related("control")
    }

    opciones = [
        ("IN_PLACE", "In Place"),
        ("CCW",      "In Place with CCW"),
        ("NA",       "Not Applicable"),
        ("NO",       "Not in Place"),
    ]

    return render(request, "users/saq_dinamico.html", {
        "requisitos":      requisitos_dict.values(),
        "saq_tipo":        tipo_codigo,
        "respuestas_dict": respuestas_dict,
        "opciones":        opciones,
    })


# ─── SAQ RESULTADO ───────────────────────────────────

@login_required
def saq_resultado(request):
    entidad = Entidad.objects.filter(usuario=request.user, is_active=True).first()
    if not entidad:
        return redirect("dashboard")

    respuestas = RespuestaSAQ.objects.filter(
        usuario=request.user, entidad=entidad
    ).select_related("control")

    total    = respuestas.count()
    no_cumple = respuestas.filter(respuesta="NO").count()
    ccw      = respuestas.filter(respuesta="CCW").count()
    na       = respuestas.filter(respuesta="NA").count()
    cumple   = respuestas.filter(respuesta="IN_PLACE").count()

    if no_cumple > 0:
        estado = "NON-COMPLIANT"
    elif ccw > 0:
        estado = "PARTIAL (CCW)"
    else:
        estado = "COMPLIANT"

    porcentaje = int((cumple / total) * 100) if total > 0 else 0

    resumen_requisitos = {}
    for r in respuestas:
        req = r.control.requisito
        if req.codigo not in resumen_requisitos:
            resumen_requisitos[req.codigo] = {"total": 0, "no": 0, "ccw": 0, "ok": 0}
        resumen_requisitos[req.codigo]["total"] += 1
        if r.respuesta == "NO":
            resumen_requisitos[req.codigo]["no"] += 1
        elif r.respuesta == "CCW":
            resumen_requisitos[req.codigo]["ccw"] += 1
        elif r.respuesta == "IN_PLACE":
            resumen_requisitos[req.codigo]["ok"] += 1

    return render(request, "users/saq_resultado.html", {
        "estado": estado, "porcentaje": porcentaje,
        "total": total, "cumple": cumple, "no_cumple": no_cumple,
        "ccw": ccw, "na": na,
        "resumen_requisitos": resumen_requisitos,
        "respuestas": respuestas,
    })

@login_required
def pretest_resultado(request):
    respuestas = RespuestaPreTest.objects.filter(usuario=request.user)

    perfil = {k: False for k in [
        "ecommerce","presencial","moto","almacena","procesa",
        "transmite","red","aislada","tercero","iframe","app"
    ]}

    mapping = {
        1:"ecommerce", 2:"presencial", 3:"moto", 4:"almacena",
        5:"procesa",   6:"transmite",  7:"red",  8:"aislada",
        9:"tercero",  10:"iframe",    11:"app",
    }

    for r in respuestas:
        if r.respuesta_opcion == "SI":
            key = mapping.get(r.pregunta.numero)
            if key:
                perfil[key] = True

    if perfil["almacena"]:
        recomendacion = "D"
    elif perfil["ecommerce"]:
        if perfil["tercero"] and perfil["iframe"]:
            recomendacion = "A"
        elif perfil["iframe"]:
            recomendacion = "A-EP"
        elif perfil["procesa"] or perfil["transmite"]:
            recomendacion = "D"
        else:
            recomendacion = "A"
    elif perfil["presencial"]:
        if perfil["aislada"]:
            recomendacion = "B"
        elif perfil["red"]:
            recomendacion = "B-IP"
        else:
            recomendacion = "C"
    elif perfil["moto"]:
        recomendacion = "C-VT"
    else:
        recomendacion = "C"

    request.session["saq_recomendado"] = recomendacion

    entidad = Entidad.objects.filter(usuario=request.user, is_active=True).first()
    if entidad:
        tipo_saq = TipoSAQ.objects.filter(codigo=recomendacion).first()
        if tipo_saq:
            prellenar_saq(request.user, entidad, tipo_saq, perfil)

    return render(request, "users/pretest_resultado.html", {
        "recomendacion": recomendacion,
        "perfil": perfil,
    })

def _calc_resumen_desde_saq(usuario, entidad, tipo_saq):
    """
    Calcula el resumen por requisito para Part 2g a partir de RespuestaSAQ.
    Retorna dict: {"1": "IN_PLACE"|"CCW"|"NA"|"NO", ...}
    """
    controles = ControlPCI.objects.filter(
        saq_links__tipo_saq=tipo_saq
    ).select_related("requisito").distinct()

    respuestas = {
        r.control_id: r.respuesta
        for r in RespuestaSAQ.objects.filter(usuario=usuario, entidad=entidad,
                                             control__in=controles)
    }

    req_status = {}
    for c in controles:
        codigo_req = c.requisito.codigo
        resp = respuestas.get(c.id, "NA")
        if codigo_req not in req_status:
            req_status[codigo_req] = {"IN_PLACE": 0, "CCW": 0, "NA": 0, "NO": 0}
        req_status[codigo_req][resp] = req_status[codigo_req].get(resp, 0) + 1

    resumen = {}
    for req, counts in req_status.items():
        if counts["NO"] > 0:
            resumen[req] = "NO"
        elif counts["CCW"] > 0:
            resumen[req] = "CCW"
        elif counts["IN_PLACE"] > 0:
            resumen[req] = "IN_PLACE"
        else:
            resumen[req] = "NA"

    return resumen

    elif paso == 5:
        resumen = {}
        for key, value in data.items():
            if key.startswith("req_"):
                req_codigo = key.replace("req_", "")
                resumen[req_codigo] = value
        aoc.resumen_requisitos = resumen
        aoc.justificacion_na   = data.get("justificacion_na", "")

    elif paso == 6:
        aoc.elegibilidad_confirmada = "elegibilidad_confirmada" in data

    elif paso == 7:
        aoc.fecha_completado    = data.get("fecha_completado") or None
        aoc.restriccion_legal   = "restriccion_legal" in data
        aoc.es_compliant        = "es_compliant" in data
        aoc.assessment_completo = data.get("assessment_tipo") != "PARTIAL"
        aoc.ack_saq_completado       = "ack_saq_completado" in data
        aoc.ack_informacion_correcta = "ack_informacion_correcta" in data
        aoc.ack_controles_mantenidos = "ack_controles_mantenidos" in data
        aoc.ejecutivo_nombre = data.get("ejecutivo_nombre", "")[:100]
        aoc.ejecutivo_cargo  = data.get("ejecutivo_cargo", "")[:100]
        aoc.fecha_firma      = data.get("fecha_firma") or None
        plan = {}
        for key, value in data.items():
            if key.startswith("plan_"):
                req_codigo = key.replace("plan_", "")
                plan[req_codigo] = (value == "SI")
        aoc.plan_accion = plan

    aoc.save()

@login_required
def aoc_wizard(request):
    entidad = Entidad.objects.filter(usuario=request.user, is_active=True).first()
    if not entidad:
        return redirect("entidades_lista")

    tipo_codigo = request.session.get("saq_recomendado", "A")
    tipo_saq    = TipoSAQ.objects.filter(codigo=tipo_codigo).first()

    # Crear o recuperar el AOCDocumento
    aoc, creado = AOCDocumento.objects.get_or_create(
        usuario=request.user,
        entidad=entidad,
        defaults={
            "tipo_saq":        tipo_saq,
            "company_name":    entidad.nombre_empresa,
            "dba":             entidad.dba,
            "contacto_nombre": entidad.contacto,
            "contacto_email":  entidad.email,
            "sitio_web_aoc":   entidad.sitio_web or "",
            "fecha_completado": date.today(),
            "fecha_firma":      date.today(),
        }
    )

    paso = int(request.GET.get("paso", 1))
    paso = max(1, min(paso, TOTAL_PASOS))

    if request.method == "POST":
        paso_actual = int(request.POST.get("paso", 1))
        _guardar_paso(aoc, paso_actual, request.POST)

        registrar_log(request, "AOC", "AOC",
                      f"AOC paso {paso_actual} guardado - Entidad: {entidad}")

        if paso_actual >= TOTAL_PASOS:
            return redirect("aoc_home")

        return redirect(f"{request.path}?paso={paso_actual + 1}")

    # ── Contexto para el paso actual ──────────────────────

    # Requisitos que aplican a este tipo de SAQ (para pasos 5 y 7)
    requisitos_aplicables = list(
        RequisitoPCI.objects.filter(
            controles__saq_links__tipo_saq=tipo_saq
        ).values("codigo", "nombre").distinct().order_by("codigo")
    )

    # Auto-calcular resumen si aún está vacío (primer acceso al paso 5)
    if paso == 5 and not aoc.resumen_requisitos:
        aoc.resumen_requisitos = _calc_resumen_desde_saq(
            request.user, entidad, tipo_saq
        )
        aoc.save()

    elegibilidad_items = ELEGIBILIDAD_POR_SAQ.get(tipo_codigo, [])

    return render(request, "users/aoc_wizard.html", {
        "aoc":                   aoc,
        "entidad":               entidad,
        "tipo_saq":              tipo_saq,
        "paso":                  paso,
        "total_pasos":           TOTAL_PASOS,
        "requisitos_aplicables": requisitos_aplicables,
        "elegibilidad_items":    elegibilidad_items,
        "hosting_opciones":      HOSTING_OPCIONES,
        "managed_opciones":      MANAGED_OPCIONES,
        "processing_opciones":   PROCESSING_OPCIONES,
        "instalaciones_json":    json.dumps(aoc.instalaciones),
        "productos_json":        json.dumps(aoc.productos_validados),
        "proveedores_json":      json.dumps(aoc.proveedores),
        "canales_json":          json.dumps(aoc.descripcion_canales),
    })
